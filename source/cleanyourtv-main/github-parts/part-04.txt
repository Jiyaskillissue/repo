    private static List<RaceWeekend.Race> raceCalendar(JSONObject json) {
        List<RaceWeekend.Race> races = new ArrayList<>();
        JSONArray list = json.optJSONArray("race_calendar");
        if (list == null) return races;
        for (int i = 0; i < list.length(); i++) {
            JSONObject r = list.optJSONObject(i);
            JSONArray sessions = r == null ? null : r.optJSONArray("sessions");
            if (sessions == null) continue;
            List<RaceWeekend.Session> parsed = new ArrayList<>();
            for (int j = 0; j < sessions.length(); j++) {
                JSONObject s = sessions.optJSONObject(j);
                if (s == null) continue;
                String name = s.optString("name", "").trim();
                try {
                    if (!name.isEmpty()) parsed.add(new RaceWeekend.Session(name, java.time.Instant.parse(s.optString("start", ""))));
                } catch (java.time.format.DateTimeParseException ignored) {
                    // a session without a readable start time cannot be counted down
                }
            }
            if (!parsed.isEmpty()) races.add(new RaceWeekend.Race(r.optInt("round", 0),
                    r.optString("name", "Grand Prix").trim(), r.optString("place", "").trim(), parsed));
        }
        return races;

    private static String revisionToken(File file) {

        try (InputStream in = new FileInputStream(file)) {
            byte[] bytes = new byte[(int) file.length()];
            int n = in.read(bytes);

    /** Reads and parses the overlay; any problem means "no overlay", never a crash on the home screen. */
    /**
     * The overlay, or before external storage is mounted, the copy kept from the last time it was read. Home can
     * start before the card is mounted (seen at every boot on a Xiaomi MiTV-MTEQ0), and without the copy it comes
     * up with the built-in dock order. A missing file on mounted storage still means no overlay.
     */
    private static JSONObject readOverlay(Context ctx, File file, boolean mounted) {
        File copy = new File(ctx.getFilesDir(), FILE_NAME);
        if (!mounted) return read(copy);
        JSONObject json = read(file);
        if (json == null) {
            copy.delete();
        } else {
            try (FileOutputStream out = new FileOutputStream(copy)) {
                out.write(json.toString().getBytes(StandardCharsets.UTF_8));
            } catch (IOException e) {
                Log.w(TAG, "could not keep a copy of " + file + ": " + e.getMessage());
            }
        }
        return json;
    }

    private static JSONObject read(File file) {
        if (file == null || !file.isFile()) return null;
        if (file.length() > 64 * 1024) {
            Log.w(TAG, "ignoring " + file + ": larger than 64 KB");
            return null;
        }
        try (InputStream in = new FileInputStream(file)) {
            byte[] buf = new byte[(int) file.length()];
            int n = 0, r;
            while (n < buf.length && (r = in.read(buf, n, buf.length - n)) > 0) n += r;
            String text = new String(buf, 0, n, StandardCharsets.UTF_8);
            if (text.startsWith("\uFEFF")) text = text.substring(1);   // Windows editors add a BOM
            return new JSONObject(text);
        } catch (Exception e) {
            Log.w(TAG, "ignoring " + file + ": " + e.getMessage());
            return null;
        }
    }

        Object v = o.opt(key);
        if (!(v instanceof String)) return fallback;
        try {
            int c = Color.parseColor(((String) v).trim());
            return c | 0xFF000000;   // always opaque: the window behind is black anyway
        } catch (IllegalArgumentException e) {
            return fallback;
        }
    }

    /** Positive keycodes only (vendor remotes use codes in the thousands); a list with anything else in it is ignored as a whole. */
    private static int[] intArray(JSONObject o, String key, int[] fallback) {
        Object v = o.opt(key);
        if (!(v instanceof JSONArray)) return fallback;
        JSONArray a = (JSONArray) v;
        int[] out = new int[a.length()];
        for (int i = 0; i < a.length(); i++) {
            Object item = a.opt(i);
            if (!(item instanceof Number)) return fallback;
            out[i] = ((Number) item).intValue();
            if (out[i] < 1) return fallback;
        }
        return out;
    }

    /** "package=value" entries; an entry without both halves is skipped, and a later entry for a package wins. */
    static Map<String, String> packageMap(List<String> entries) {
        Map<String, String> out = new HashMap<>();
        for (String entry : entries) {
            int eq = entry.indexOf('=');
            if (eq <= 0) continue;
            String pkg = entry.substring(0, eq).trim(), value = entry.substring(eq + 1).trim();
            if (!pkg.isEmpty() && !value.isEmpty()) out.put(pkg, value);
        }
        return out;
    }

    private static List<String> stringList(JSONObject o, String key, List<String> fallback) {
        Object v = o.opt(key);
        if (!(v instanceof JSONArray)) return fallback;
        JSONArray a = (JSONArray) v;
        List<String> out = new ArrayList<>();
        for (int i = 0; i < a.length(); i++) {
            Object item = a.opt(i);
            if (item instanceof String && !((String) item).trim().isEmpty()) out.add(((String) item).trim());
        }
        return out;
    }

/**
 * Inputs on Coocaa OS (Skyworth) TVs, which register nothing with TvInputManager. com.skyworth.tv lists
 * the sources its own Source panel shows (HDMI 1, HDMI 2, RF, AV), and switches to one and brings up its
 * player, as picking it in that panel does.
 */
final class CoocaaSources {

    private static final Uri LIST = Uri.parse("content://com.skyworth.tv.provider/system/source_list");
    private static final Uri METHOD = Uri.parse("content://com.skyworth.tv.sdk.provider/method");
    private static final String[] COLUMNS = {"sourceName", "sourceIndex", "displayName"};

    static final class Source {
        /** HDMI, AV, VOD (the RF tuner), ... */
        final String name;
        final int index;
        final String label;

        Source(String name, int index, String label) {
            this.name = name;
            this.index = index;
            this.label = label;
        }
    }

    private CoocaaSources() {
    }

    /** HDMI ports in port order, then AV, then the tuner; empty where the provider is missing. */
    static List<Source> list(Context ctx) {
        List<Source> out = new ArrayList<>();
        try (Cursor c = ctx.getContentResolver().query(LIST, COLUMNS, null, null, null)) {
            while (c != null && c.moveToNext()) {
                String name = c.getString(0);
                String label = c.getString(2);
                if (name == null) continue;
                out.add(new Source(name, c.getInt(1), label == null || label.isEmpty() ? name : label));
            }
        } catch (RuntimeException e) {
            return Collections.emptyList();
        }
        Collections.sort(out, (a, b) -> a.name.equals(b.name) ? Integer.compare(a.index, b.index)
                : Integer.compare(rank(a.name), rank(b.name)));
        return out;
    }

    private static int rank(String name) {
        switch (name) {
            case "HDMI": return 0;
            case "AV": return 1;
            case "VOD": return 3;
            default: return 2;
        }
    }

    static boolean open(Context ctx, Source source) {
        Bundle extras = new Bundle();
        extras.putString("name", source.name);
        extras.putInt("index", source.index);
        try {
            Bundle ret = ctx.getContentResolver().call(METHOD, "METHOD_GOTO_SOURCE", null, extras);
            return ret != null && ret.getBoolean("ret");
        } catch (RuntimeException e) {
            return false;
        }
    }
}

    private float noise(float x) {
        int i = (int) Math.floor(x);
        float f = x - i;
        float s = f * f * (3f - 2f * f);
        float a = noise[i & 255], b = noise[(i + 1) & 255];
        return a + (b - a) * s;
    }

    /** sin of a phase given in turns. */
    private static float sin(float turns) {
        return (float) Math.sin(turns * 2.0 * Math.PI);
    }

    private static float clamp01(float v) {
        return v < 0f ? 0f : (v > 1f ? 1f : v);
    }

    private static float decel(float p) {
        float q = 1f - p;
        return 1f - q * q;
    }

    private static float backOut(float x) {
        float s = 1.4f, q = x - 1f;
        return 1f + (s + 1f) * q * q * q + s * q * q;
    }

        Paint p = new Paint(Paint.ANTI_ALIAS_FLAG);

import java.util.ArrayList;
import java.util.List;
import java.util.Set;

/**
 * The dock is the first row of the grid. dock_apps decides what is in it, or the saved order once the owner
 * changed the dock on the TV: a slot left free stays empty instead of taking the first grid app (#162) when
 * fill_dock is false; by default they fill.
 */
final class DockSlots {
    private DockSlots() { }

    /**
     * Empty slots to put after the {@code docked} dock apps found. None when the dock is full, when no
     * dock app is installed (the first row fills so Home still has a dock) or when nothing follows.
     */
    static int empty(int docked, int dockSize, int tiles, boolean fill) {
        if (fill || docked <= 0 || docked >= dockSize || tiles <= docked) return 0;
        return dockSize - docked;
    }

    /**
     * The order a reorder session saves when fill_dock is false (#499). Reorder mode shows no empty slots, so a
     * grid app slides into a free dock slot without being put there. Only the keys in {@code dock} (in the dock
     * when the session started, or carried into it) that end inside the first {@code dockSize} places stay in the
     * dock: they come first, in their order, and every other key follows in its order. The dock size to save is
     * the number of keys in {@code dock} found there.
     */
    static List<String> dockFirst(List<String> keys, int dockSize, Set<String> dock) {
        List<String> head = new ArrayList<>(), tail = new ArrayList<>();
        for (int i = 0; i < keys.size(); i++) {
            String k = keys.get(i);
            (i < dockSize && dock.contains(k) ? head : tail).add(k);
        }
        head.addAll(tail);
        return head;
    }

    /** How many of the first {@code dockSize} keys are in {@code dock}: the dock size dockFirst puts first. */
    static int dockCount(List<String> keys, int dockSize, Set<String> dock) {
        int n = 0;
        for (int i = 0; i < Math.min(dockSize, keys.size()); i++) if (dock.contains(keys.get(i))) n++;
        return n;
    }

    /** Whether a tile sits in the dock row: every tile of a single row, else the first `dockSize` of the grid. */
    static boolean inDockRow(boolean dockRow, boolean singleRow, int position, int dockSize) {
        return dockRow && (singleRow || position < dockSize);
    }

    /** A position focus can land on: an empty slot sends it to the last dock app. */
    static int focusable(int position, int docked, int empty) {
        return empty > 0 && position >= docked && position < docked + empty ? docked - 1 : position;
    }
}

import android.content.res.Resources;

import java.time.LocalDate;
import java.time.format.DateTimeParseException;

/**
 * A festive skin set by a theme's config.json: {@code festival} (or {@code seasonal_skin}), {@code festival_until},
 * {@code greeting} and {@code accent_color}. While it is active the pills get festive icons (a diya for the
 * network, a bell for Inputs, a marigold for Settings; for {@code halloween} a jack-o'-lantern, a bat and a
 * ghost), the greeting is replaced and the focus glow takes the accent colour. It switches itself off after
 * {@code festival_until}, so nobody has to remember to undo it.
 */
final class Festival {

    static final Festival NONE = new Festival("", null, "", 0);

    /** Warm diya gold, the accent when a theme names a festival but no colour. */
    static final int DEFAULT_ACCENT = 0xFFFFB23C;

    /** The festival with its own skin (jack-o'-lantern, bat, ghost, bats on the first Home) instead of the diya's. */
    static final String HALLOWEEN = "halloween";

    /** Pumpkin orange, Halloween's accent when the theme names no colour. */
    static final int PUMPKIN = 0xFFFF7A1A;

    /** Lower-case festival name, or "" for none. */
    final String name;
    /** Last day the skin shows, inclusive; null for no end date. */
    final LocalDate until;
    /** Text that replaces the time-of-day greeting; "" uses the festival's own default. */
    final String greeting;
    /** Focus glow colour as 0xAARRGGBB, or 0 for {@link #DEFAULT_ACCENT}. */
    final int accent;

    private Festival(String name, LocalDate until, String greeting, int accent) {
        this.name = name;
        this.until = until;
        this.greeting = greeting;
        this.accent = accent;
    }

    /**
     * Builds the skin from raw config values. Anything malformed turns the skin off rather than leaving a
     * festival on the screen forever: an unreadable end date or an empty name gives {@link #NONE}.
     */
    static Festival parse(String name, String until, String greeting, String accent) {
        String n = name == null ? "" : name.trim().toLowerCase(Locale.ROOT);
        if (n.isEmpty() || n.length() > 40) return NONE;
        LocalDate end = null;
        if (until != null && !until.trim().isEmpty()) {
            try {
                end = LocalDate.parse(until.trim());
            } catch (DateTimeParseException e) {
                return NONE;
            }
        }
        String g = greeting == null ? "" : Cards.cleanLabel(greeting, "");
        return new Festival(n, end, g, parseColor(accent));
    }

    /** Whether the skin shows today. */
    boolean active(LocalDate today) {
        return !name.isEmpty() && (until == null || !today.isAfter(until));
    }

    /** The greeting to show while active, or "" to keep the normal time-of-day greeting. */
    String greetingText(Resources res) {
        if (!greeting.isEmpty()) return greeting;
        switch (name) {
            case "dussehra":
                return res.getString(R.string.festival_dussehra_greeting);
            case "diwali":
                return res.getString(R.string.festival_diwali_greeting);
            case HALLOWEEN:
                return res.getString(R.string.festival_halloween_greeting);
            default:
                return "";
        }
    }

    int accentColor() {
        if (accent != 0) return accent;
        return isHalloween() ? PUMPKIN : DEFAULT_ACCENT;
    }

    /** Whether this is the Halloween skin rather than the diya, bell and marigold. */
    boolean isHalloween() {
        return HALLOWEEN.equals(name);
    }

    /** {@code #RRGGBB} to an opaque colour, anything else to 0. */
    static int parseColor(String s) {
        if (s == null) return 0;
        String t = s.trim();
        if (!t.matches("#[0-9A-Fa-f]{6}")) return 0;
        return 0xFF000000 | Integer.parseInt(t.substring(1), 16);
    }
}

/**
 * game_day (config.xml or a theme): each team's next game in the race panel's place, from an add-on such as
 * Game Day, rotating with the theme's F1 weekend. The launcher only reads what the add-on publishes and stays
 * offline. Contract: docs/modules.md.
 */
final class GameDay {
    static final String META = "com.gathin.tvhome.games";
    static final int TURN_DEFAULT = 30;
    private static final String[] COLUMNS = {"top_line", "word", "start_ms", "end_ms", "colour", "calendar", "turn_seconds"};

    private final Context ctx;
    private final Uri uri;
    private final Handler main;
    private final Runnable onChange;
    private final ExecutorService io = Executors.newSingleThreadExecutor();
    private final ContentObserver observer;
    private boolean watching;
    private List<Turn> turns = Collections.emptyList();
    private int turnSeconds = TURN_DEFAULT;

    private GameDay(Context ctx, Uri uri, Handler main, Runnable onChange) {
        this.ctx = ctx;
        this.uri = uri;
        this.main = main;
        this.onChange = onChange;
        observer = new ContentObserver(main) {
            @Override
            public void onChange(boolean selfChange) {
                read();
            }
        };
    }

    /**
     * Null when the key is empty, the package is not installed, or it names no provider of its own in a
     * {@value #META} meta-data. Naming the package is the owner's opt-in.
     */
    static GameDay create(Context ctx, String pkg, Handler main, Runnable onChange) {
        if (pkg == null || pkg.trim().isEmpty()) return null;
        try {
            PackageManager pm = ctx.getPackageManager();
            ApplicationInfo info = pm.getApplicationInfo(pkg.trim(), PackageManager.GET_META_DATA);
            String authority = info.metaData == null ? null : info.metaData.getString(META);
            if (authority == null || authority.trim().isEmpty()) return null;
            // The provider must be the named package's own, not another app's that claims the authority.
            ProviderInfo provider = pm.resolveContentProvider(authority.trim(), 0);
            if (provider == null || !pkg.trim().equals(provider.packageName)) return null;
            Uri uri = new Uri.Builder().scheme(ContentResolver.SCHEME_CONTENT).authority(authority.trim()).appendPath("turns").build();
            return new GameDay(ctx, uri, main, onChange);
        } catch (PackageManager.NameNotFoundException e) {
            return null;
        }
    }

    /** Home is on screen: read now and watch for changes. */
    void start() {
        if (!watching) {
            try {
                ctx.getContentResolver().registerContentObserver(uri, false, observer);
                watching = true;
            } catch (SecurityException e) {
                // The add-on's provider went away; the panel keeps what it had.
            }
        }
        read();
    }

    /** Home is off screen: nothing runs until it is back. */
    void stop() {
        if (watching) ctx.getContentResolver().unregisterContentObserver(observer);
        watching = false;
    }

    void shutdown() {
        stop();
        io.shutdownNow();
    }

    void read() {

            io.execute(() -> {
                List<Object[]> rows = new ArrayList<>();

                    while (c != null && c.moveToNext()) {
                        rows.add(new Object[] {c.getString(0), c.getString(1), c.getLong(2), c.getLong(3), c.getInt(4),

                    return;

                    turnSeconds = seconds;

                });
            });

            // Home is being destroyed.
        }
    }

        return turns;
    }

        return turnSeconds;
    }

    /** The add-on's rows as turns; a row without a top line or with impossible times is skipped. */
    static List<Turn> fromRows(List<Object[]> rows) {
        List<Turn> out = new ArrayList<>();
        for (Object[] r : rows) {
            if (!(r[0] instanceof String) || ((String) r[0]).isEmpty()) continue;
            long start = r[2] instanceof Number ? ((Number) r[2]).longValue() : -1;
            long end = r[3] instanceof Number ? ((Number) r[3]).longValue() : -1;

            int colour = r[4] instanceof Number ? ((Number) r[4]).intValue() : 0xFFFFFFFF;
            out.add(new Turn((String) r[0], word, start, end, colour));
        }
        return out;
    }

    static int turnSeconds(List<Object[]> rows) {
        if (rows.isEmpty() || !(rows.get(0)[6] instanceof Number)) return TURN_DEFAULT;
        int s = ((Number) rows.get(0)[6]).intValue();
        return s >= 10 && s <= 120 ? s : TURN_DEFAULT;
    }

        final String topLine;
        /** "Puck drop": the countdown reads "Puck drop in 2h 18m". */
        final String word;
        /** For the F1 turn: when its next session starts, or now while one is live (the sort key). */
        final long startMs;
        final long endMs;
        /** ARGB, for the short bar. */
        final int colour;
        final boolean f1;

        Turn(String topLine, String word, long startMs, long endMs, int colour) {
            this(topLine, word, startMs, endMs, colour, false);

        private Turn(String topLine, String word, long startMs, long endMs, int colour, boolean f1) {

            this.colour = colour;
            this.f1 = f1;
        }
    }

     * while the race panel would show it (sorted by its next session, or now while a session is live).
     */
    static List<Turn> merge(List<Turn> games, RaceWeekend.State f1, long nowMs) {
        List<Turn> out = new ArrayList<>();
        for (Turn t : games) {
            if (t.endMs > nowMs) out.add(t);
        }
        if (f1 != null) {
            long key = f1.live == null && f1.next != null ? f1.next.start.toEpochMilli() : nowMs;
            out.add(new Turn("", "", key, Long.MAX_VALUE, 0, true));
        }
        Collections.sort(out, Comparator.comparingLong(t -> t.startMs));
        return out;
    }

    /** Which turn is on screen, {@code sinceMs} after the rotation started. */
    static int current(int count, long sinceMs, int turnSeconds) {
        if (count <= 0) return 0;
        return (int) ((Math.max(0, sinceMs) / (turnSeconds * 1000L)) % count);
    }

    /** Milliseconds until the next turn, {@code sinceMs} after the rotation started. */
    static long delayToNextTurn(long sinceMs, int turnSeconds) {
        long turnMs = turnSeconds * 1000L;
        return turnMs - Math.floorMod(sinceMs, turnMs);
    }

    static boolean live(Turn t, long nowMs) {
        return nowMs >= t.startMs && nowMs < t.endMs;
    }

    static Duration until(Turn t, long nowMs) {
        return Duration.ofMillis(t.startMs - nowMs);
    }
