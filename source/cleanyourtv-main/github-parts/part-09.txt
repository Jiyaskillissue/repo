    /**
     * Hold OK on a tile: a glass menu next to it. Move starts reorder mode with this tile already picked up;
     * Remove from Home takes the shortcut off (only with the app library on, where one press adds it back);
     * App info opens the TV's page for the app, which is what holding OK used to do.
     */
    // ---- hold OK ---------------------------------------------------------------------------------
    // View's own long click misses on some remotes: an IR remote that sends its repeat frames as separate
    // presses, or a Bluetooth one whose repeats land after a tile rebind, never holds one DOWN long enough.
    // So the launcher times the hold itself: OK down on a tile, still down (or still repeating) after
    // HOLD_OK_MS opens the tile menu, and the UP that follows is eaten so it doesn't also open the app.

    private static final long HOLD_OK_MS = 550;
    private long holdOkDown = -1;
    private int holdOkPosition = -1;
    private String holdOkKey;    // the app under OK when it went down; a tile rebuild can move another app to that slot
    private boolean holdOkFired;
    private final Runnable holdOkTimer = this::fireHoldOk;

    private static boolean isOkKey(int code) {
        return code == KeyEvent.KEYCODE_DPAD_CENTER || code == KeyEvent.KEYCODE_ENTER
                || code == KeyEvent.KEYCODE_NUMPAD_ENTER || code == KeyEvent.KEYCODE_BUTTON_A;
    }

    /** True when the event belongs to a hold that opened the tile menu and must not reach the tile. */
    private boolean holdOk(KeyEvent event) {
        if (!tileMenuEnabled || !isOkKey(event.getKeyCode())) return false;
        if (event.getAction() == KeyEvent.ACTION_DOWN) {
            if (event.getRepeatCount() == 0) {
                holdOkFired = false;
                handler.removeCallbacks(holdOkTimer);
                holdOkPosition = reordering || openPopup != null || cardPrompt != null ? -1 : focusedTilePosition();
                holdOkKey = holdOkPosition < 0 || holdOkPosition >= adapter.appCount() ? null : adapter.keyAt(holdOkPosition);
                if (holdOkKey == null) {
                    holdOkDown = -1;
                    return false;
                }
                holdOkDown = event.getDownTime();
                handler.postDelayed(holdOkTimer, HOLD_OK_MS);
                return false;
            }
            if (holdOkFired) return true;
            if (holdOkDown >= 0 && event.getEventTime() - holdOkDown >= HOLD_OK_MS) {
                fireHoldOk();
                return true;
            }
            return false;
        }
        // ACTION_UP
        handler.removeCallbacks(holdOkTimer);
        holdOkDown = -1;
        holdOkKey = null;
        if (holdOkFired) {
            holdOkFired = false;
            View f = getCurrentFocus();
            if (f != null) f.setPressed(false);
            return true;
        }
        return false;
    }

    /**
     * Stops timing a held OK when the launcher is paused, loses its window or goes away, so the menu can't open
     * behind another screen. An interrupted press still counts as handled: its UP, if it comes back here, is eaten
     * rather than opening the tile.
     */
    private void cancelHoldOk() {
        handler.removeCallbacks(holdOkTimer);
        if (holdOkDown >= 0) holdOkFired = true;
        holdOkDown = -1;
        holdOkPosition = -1;
        holdOkKey = null;
    }

    private void fireHoldOk() {
        handler.removeCallbacks(holdOkTimer);
        if (holdOkDown < 0 || holdOkFired) return;
        holdOkFired = true;
        if (openPopup != null || reordering) return;   // View's own long click got there first
        int position = holdOkPosition;
        if (position < 0 || position >= adapter.appCount() || position != focusedTilePosition()) return;
        // Same slot is not enough: tiles may have been rebuilt during the hold, so the app must match too.
        if (!holdOkKey.equals(adapter.keyAt(position))) return;
        View card = getCurrentFocus();
        if (card != null) card.setPressed(false);
        onTileLongPress(position, card);
    }

    private void buildRing(int ring, Random rnd) {
        int n = COUNT[ring];
        petals[ring] = new Path[n];
        angles[ring] = new float[n];
        colors[ring] = new int[n];
        edges[ring] = new int[n];

        float edgeDepth = ring == 0 ? 0.55f : 0.45f;

            float deg = offset + j * 360f / n + (rnd.nextFloat() - 0.5f) * 10f;
            float r = RADIUS[ring] * (0.96f + 0.04f * rnd.nextFloat());

    /** One petal: narrow at the base, fanning out to a frilled, notched edge. */
    private static Path petal(float a, float hw, float r0, float r, int scallops, Random rnd) {
        Path p = new Path();
        float a0 = a - hw, a1 = a + hw;
        p.moveTo(px(r0, a - hw * 0.35f), py(r0, a - hw * 0.35f));
        float sideA = a0 - hw * 0.08f;
        p.quadTo(px(r * 0.62f, sideA), py(r * 0.62f, sideA), px(r * 0.86f, a0), py(r * 0.86f, a0));
        int steps = scallops * 2;
        for (int k = 1; k < steps; k += 2) {
            float ta = a0 + (a1 - a0) * k / steps + (rnd.nextFloat() - 0.5f) * hw * 0.12f;
            float na = a0 + (a1 - a0) * (k + 1) / steps;
            float tr = r * (1.0f + 0.06f * rnd.nextFloat());
            float nr = k + 1 >= steps ? r * 0.86f : r * (0.83f + 0.06f * rnd.nextFloat());
            p.quadTo(px(tr, ta), py(tr, ta), px(nr, na), py(nr, na));
        }
        float sideB = a1 + hw * 0.08f;
        float endA = a + hw * 0.35f;
        p.quadTo(px(r * 0.62f, sideB), py(r * 0.62f, sideB), px(r0, endA), py(r0, endA));
        p.close();
        return p;
    }

    private void buildCore(Random rnd) {
        int bumps = 18;
        for (int k = 0; k <= bumps * 2; k++) {
            float a = (float) (Math.PI * k / bumps);
            float r = (k % 2 == 1) ? CORE_R * (1.0f + 0.06f * rnd.nextFloat()) : CORE_R * 0.84f;
            if (k == 0) core.moveTo(px(r, a), py(r, a));
            else if (k % 2 == 1) {
                float na = (float) (Math.PI * (k + 1) / bumps);
                float nr = CORE_R * 0.84f;
                core.quadTo(px(r, a), py(r, a), px(nr, na), py(nr, na));
            }
        }
        core.close();
        // tiny florets in a sunflower spiral
        double golden = Math.PI * (3 - Math.sqrt(5));
        for (int k = 0; k < seeds.length / 2; k++) {

            float a = (float) (k * golden);
            seeds[k * 2] = px(r, a);
            seeds[k * 2 + 1] = py(r, a);
        }
    }

/** show_stats' memory figure from the lines of /proc/meminfo. */
final class MemInfo {
    private MemInfo() {}

    /**
     * RAM usage as {used, total} in MB, used = MemTotal - MemAvailable, e.g. "945 / 1536 MB". Used over free so
     * the line reads as a load against a known whole. MemAvailable only exists since Linux 3.14; on older kernels
     * (TPM171E Philips, Android TV 8, kernel 3.10) available is MemFree + Buffers + Cached, the estimate free(1)
     * used before it. Null when MemTotal or every available figure is missing.
     */
    static long[] usedOfTotalMb(Iterable<String> lines) {
        long total = -1, available = -1, free = -1, buffers = -1, cached = -1;
        for (String line : lines) {
            if (line.startsWith("MemTotal:")) total = kb(line);
            else if (line.startsWith("MemAvailable:")) available = kb(line);
            else if (line.startsWith("MemFree:")) free = kb(line);
            else if (line.startsWith("Buffers:")) buffers = kb(line);
            else if (line.startsWith("Cached:")) cached = kb(line);
        }
        if (available < 0 && free >= 0) available = free + Math.max(0, buffers) + Math.max(0, cached);
        if (available < 0 || total <= 0) return null;
        long totalMb = total / 1024;
        long availableMb = Math.min(total, available) / 1024;
        return new long[]{Math.max(0, totalMb - availableMb), totalMb};
    }

    /** The kB figure of a "Name:   1234 kB" line, or -1. */
    private static long kb(String line) {
        String[] t = line.trim().split("\\s+");
        if (t.length < 2) return -1;
        try {
            return Long.parseLong(t[1]);
        } catch (NumberFormatException e) {
            return -1;
        }
    }

import android.content.Context;
import android.util.AtomicFile;
import android.util.Log;

import java.io.File;

/**
 * The owner's own tile order, set with the reorder button and kept in the launcher's private files (order.txt).
 *
 * Saved tiles come first in this order; apps not in it (new installs) follow by the usual rule (dock_apps, cards,
 * alphabetical) and uninstalled ones simply drop out. The file also records the dock_apps it was made under: when
 * dock_apps changes (a rebuild or a pushed theme), the saved order is ignored, so the new dock takes effect until
 * the owner reorders again.
 *
 * Format, UTF-8 text: {@code tvhome-order 1}, then {@code dock <comma-separated dock_apps>}, then optionally
 * {@code docked <n>}, then one tile key (component, flattened short) per line. {@code docked} says how many of the
 * first keys are in the dock, for fill_dock false, where an app the owner moved into the dock is not in dock_apps
 * (#499). Files without it (older launchers) leave the dock to dock_apps; an older launcher reads the line as a key
 * that matches no tile.
 */
final class Order {

    private static final String TAG = "TvHome.Order";
    static final String FILE_NAME = "order.txt";
    private static final String MAGIC = "tvhome-order 1";
    private static final String DOCK = "dock ";
    private static final String DOCKED = "docked ";

    /** dock_apps the order was saved under, comma-joined (package names cannot contain commas). */
    final String dock;
    final List<String> keys;
    /** How many of the first keys are in the dock, or -1 when not recorded (dock_apps decides). */
    final int docked;

    Order(String dock, List<String> keys) {
        this(dock, keys, -1);
    }

    Order(String dock, List<String> keys, int docked) {
        this.dock = dock;
        this.keys = Collections.unmodifiableList(new ArrayList<>(keys));
        this.docked = docked;
    }

    static String dockKey(List<String> dockApps) {
        return String.join(",", dockApps);
    }

    static String encode(Order order) {
        StringBuilder sb = new StringBuilder(MAGIC).append('\n').append(DOCK).append(order.dock).append('\n');
        if (order.docked >= 0) sb.append(DOCKED).append(order.docked).append('\n');
        for (String k : order.keys) sb.append(k).append('\n');
        return sb.toString();
    }

    /** The order in this text, or null when it is not one (wrong header, a blank or duplicate key). */
    static Order decode(String text) {
        if (text == null) return null;
        String[] lines = text.split("\n", -1);
        if (lines.length < 2 || !lines[0].equals(MAGIC) || !lines[1].startsWith(DOCK)) return null;
        int docked = -1, first = 2;
        if (lines.length > 2 && lines[2].startsWith(DOCKED)) {
            try {
                docked = Integer.parseInt(lines[2].substring(DOCKED.length()));
            } catch (NumberFormatException e) {
                return null;
            }
            if (docked < 0) return null;
            first = 3;
        }
        List<String> keys = new ArrayList<>();
        for (int i = first; i < lines.length; i++) {
            String k = lines[i];
            if (k.isEmpty()) {
                if (i == lines.length - 1) break;   // the trailing newline
                return null;
            }
            if (keys.contains(k)) return null;
            keys.add(k);
        }
        if (docked > keys.size()) return null;
        return new Order(lines[1].substring(DOCK.length()), keys, docked);
    }

    /**
     * Rank of each saved key (0 = first) when the order still applies to these dock_apps, else an empty map.
     * Tiles without a rank sort after every ranked one.
     */
    static Map<String, Integer> ranks(Order order, List<String> dockApps) {
        Map<String, Integer> r = new HashMap<>();
        if (order == null || !order.dock.equals(dockKey(dockApps))) return r;
        for (int i = 0; i < order.keys.size(); i++) r.put(order.keys.get(i), i);
        return r;
    }

    /** The saved dock size when the order still applies to these dock_apps and recorded one, else -1. */
    static int docked(Order order, List<String> dockApps) {
        return order == null || !order.dock.equals(dockKey(dockApps)) ? -1 : order.docked;
    }

    /** Sort step for two tiles: saved ones first in saved order, then 0 to fall through to the usual rule. */
    static int compare(Map<String, Integer> ranks, String keyA, String keyB) {
        Integer a = ranks.get(keyA), b = ranks.get(keyB);
        if (a == null && b == null) return 0;
        if (a == null) return 1;
        if (b == null) return -1;
        return Integer.compare(a, b);
    }

    // ---- file ------------------------------------------------------------------------------------

    static File file(Context ctx) {
        return new File(ctx.getFilesDir(), FILE_NAME);
    }

    /** The saved order, or null when there is none or it does not decode (then it is ignored, not deleted). */
    static Order load(File file) {
        if (!file.exists()) return null;
        try {
            Order o = decode(new String(new AtomicFile(file).readFully(), StandardCharsets.UTF_8));
            if (o == null) Log.w(TAG, file.getName() + " is not valid; ignored");
            return o;
        } catch (IOException e) {
            Log.w(TAG, "cannot read " + file.getName() + ": " + e);
            return null;
        }
    }

    /** Writes the order atomically. False (and a log line) when the write failed. */
    static boolean save(File file, Order order) {
        AtomicFile af = new AtomicFile(file);
        FileOutputStream out = null;
        try {
            out = af.startWrite();
            out.write(encode(order).getBytes(StandardCharsets.UTF_8));
            af.finishWrite(out);
            return true;
        } catch (IOException e) {
            Log.w(TAG, "cannot write " + file.getName() + ": " + e);
            if (out != null) af.failWrite(out);
            return false;
        }
    }

            xFrac[i] = (i + rnd.nextFloat()) / n;          // spread evenly across the width
            delay[i] = rnd.nextFloat() * 0.32f;             // fractions of the duration
            life[i] = 0.6f + rnd.nextFloat() * (1f - delay[i] - 0.6f);
            drop[i] = 0.8f + rnd.nextFloat() * 0.35f;       // fraction of the height fallen over its life
            size[i] = 9f + rnd.nextFloat() * 7f;            // dp
            swayAmp[i] = 10f + rnd.nextFloat() * 24f;       // dp
            swayFreq[i] = 0.5f + rnd.nextFloat() * 0.6f;    // Hz
            swayPhase[i] = rnd.nextFloat() * (float) TAU;
            rot0[i] = rnd.nextFloat() * 360f;
            spin[i] = (40f + rnd.nextFloat() * 120f) * (rnd.nextBoolean() ? 1f : -1f); // degrees per second
            flipFreq[i] = 0.4f + rnd.nextFloat();           // Hz
            flipPhase[i] = rnd.nextFloat() * (float) TAU;
            color[i] = rnd.nextInt(COLORS.length);
            shape[i] = rnd.nextInt(shapes.length);
        }
        playing = true;
        start = SystemClock.uptimeMillis();
        postDelayed(finish, duration);
        postInvalidateOnAnimation();
    }

    private void finish() {
        if (!playing) return;
        playing = false;
        invalidate();
        Runnable done = onDone;
        onDone = null;
        if (done != null) done.run();
    }

    @Override
    protected void onDetachedFromWindow() {
        super.onDetachedFromWindow();
        if (playing) {
            removeCallbacks(finish);
            finish();
        }
    }

    @Override
    protected void onDraw(Canvas c) {
        if (!playing) return;
        long now = SystemClock.uptimeMillis();
        float elapsed = (now - start) / (float) duration;
        float secs = (now - start) / 1000f;
        float w = getWidth(), h = getHeight();
        for (int i = 0; i < count; i++) {
            float p = (elapsed - delay[i]) / life[i];
            if (p <= 0f || p >= 1f) continue;
            float len = size[i] * density;
            float fall = (p + p * p) * 0.5f;               // gathers speed, like a petal against the air
            float y = -len * 1.5f + (h * drop[i] + len * 1.5f) * fall;
            double sway = TAU * swayFreq[i] * secs + swayPhase[i];
            float x = xFrac[i] * w + swayAmp[i] * density * (float) Math.sin(sway);
            float angle = rot0[i] + spin[i] * secs + 22f * (float) Math.cos(sway); // tilts into the sway
            float flip = (float) Math.cos(TAU * flipFreq[i] * secs + flipPhase[i]);
            float sx = Math.abs(flip) < 0.08f ? (flip < 0f ? -0.08f : 0.08f) : flip;
            int a = p > 0.75f ? (int) (255 * (1f - (p - 0.75f) / 0.25f)) : 255;

            // turned edge-on it looks darker; the back is darker still
            float shade = (flip < 0f ? 0.78f : 1f) * (0.8f + 0.2f * Math.abs(flip));
            int base = COLORS[color[i]];
            fill.setColor(scale(base, shade));
            fill.setAlpha(a);
            edge.setColor(scale(base, 0.55f));
            edge.setAlpha(a * 3 / 4);
            vein.setAlpha(a / 4);

            c.save();
            c.translate(x, y);
            c.rotate(angle);
            c.scale(len * sx, len);
            c.drawPath(shapes[shape[i]], fill);
            c.drawPath(shapes[shape[i]], edge);
            c.drawPath(veins[shape[i]], vein);
            c.restore();
        }
        postInvalidateOnAnimation();
    }

    private void ensureCapacity(int n) {
        if (xFrac.length >= n) return;
        xFrac = new float[n];
        delay = new float[n];
        life = new float[n];
        drop = new float[n];
        size = new float[n];
        swayAmp = new float[n];
        swayFreq = new float[n];
        swayPhase = new float[n];
        rot0 = new float[n];
        spin = new float[n];
        flipFreq = new float[n];
        flipPhase = new float[n];
        color = new int[n];
        shape = new int[n];
    }

    /** One marigold petal, 1 unit tall around the origin: a narrow base fanning out to a frilled top. */
    private static Path petalShape(Random r) {
        Path p = new Path();
        float wTop = 0.36f + 0.08f * r.nextFloat();
        p.moveTo(0f, 0.5f);
        p.cubicTo(-0.16f, 0.36f, -wTop - 0.04f, 0.02f, -wTop, -0.3f);
        int scallops = 4 + r.nextInt(2);
        for (int k = 0; k < scallops; k++) {
            float tx = -wTop + 2 * wTop * (k + 0.5f) / scallops;
            float nx = -wTop + 2 * wTop * (k + 1f) / scallops;
            float tipY = -0.5f + 0.2f * sq(tx / wTop) - 0.06f - 0.05f * r.nextFloat();
            float notchY = k == scallops - 1 ? -0.3f : -0.5f + 0.2f * sq(nx / wTop) + 0.05f + 0.03f * r.nextFloat();
            p.quadTo(tx, tipY, nx, notchY);
        }
        p.cubicTo(wTop + 0.04f, 0.02f, 0.16f, 0.36f, 0f, 0.5f);
        p.close();
        return p;
    }

    private static Path veinPath(Random r) {
        Path p = new Path();
        float bend = (r.nextFloat() - 0.5f) * 0.06f;
        p.moveTo(0f, 0.42f);
        p.quadTo(bend, 0f, 0f, -0.36f);
        p.moveTo(0f, 0.3f);
        p.quadTo(-0.12f, -0.05f, -0.2f, -0.3f);
        p.moveTo(0f, 0.3f);
        p.quadTo(0.12f, -0.05f, 0.2f, -0.3f);
        return p;
    }

    private static float sq(float v) {
        return v * v;
    }

    private static int scale(int c, float f) {
        int r = Math.min(255, (int) (((c >> 16) & 0xFF) * f));
        int g = Math.min(255, (int) (((c >> 8) & 0xFF) * f));
        int b = Math.min(255, (int) ((c & 0xFF) * f));
        return 0xFF000000 | (r << 16) | (g << 8) | b;
    }

import android.content.res.Resources;

import java.time.Duration;
import java.time.Instant;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Locale;

/**
 * Race-weekend mode. A theme's config.json can carry an F1 calendar ({@code race_calendar}, written by
 * tools/race_calendar.py); from shortly before a weekend's first session until the race is over, Home shows
 * a small panel with the next session and a countdown. Scheduling uses the clock against owner-pushed
 * data and is tested on the JVM; display labels use Android resources.
 */
final class RaceWeekend {

    /** The panel shows from this long before a weekend's first session... */
    static final Duration LEAD = Duration.ofHours(12);
    /** ...until this long after the race starts: a race runs about two hours, then the result settles. */
    static final Duration TAIL = Duration.ofHours(4);

    enum Status { DONE, LIVE, NEXT, LATER }

    static final class Session {
        /** As written in the calendar: FP1, FP2, FP3, SPRINT QUALI, SPRINT, QUALIFYING or RACE. */
        final String name;
        final Instant start;

        Session(String name, Instant start) {
            this.name = name.trim().toUpperCase(Locale.ROOT);
            this.start = start;
        }

        boolean isRace() {
            return name.equals("RACE");
        }

        /** Roughly how long the session is live; only used to tell live from done. */
        Duration length() {
            if (isRace()) return Duration.ofHours(2);
            if (name.equals("SPRINT QUALI")) return Duration.ofMinutes(45);
            return Duration.ofHours(1);
        }

        /** The name in the countdown line: "Qualifying in 2h 14m". */
        String title(Resources res) {
            switch (name) {
                case "FP1": return res.getString(R.string.race_practice_1);
                case "FP2": return res.getString(R.string.race_practice_2);
                case "FP3": return res.getString(R.string.race_practice_3);
                case "SPRINT QUALI": return res.getString(R.string.race_sprint_qualifying);
                case "SPRINT": return res.getString(R.string.race_sprint);
                case "QUALIFYING": return res.getString(R.string.race_qualifying);
                case "RACE": return res.getString(R.string.race_race);
                default: return name.charAt(0) + name.substring(1).toLowerCase(Locale.ROOT);
            }
        }

        /** The name in the session row, kept short so a whole weekend fits on one line. */
        String shortName(Resources res) {
            switch (name) {
                case "FP1": return res.getString(R.string.race_practice_1_short);
                case "FP2": return res.getString(R.string.race_practice_2_short);
                case "FP3": return res.getString(R.string.race_practice_3_short);
                case "SPRINT QUALI": return res.getString(R.string.race_sprint_qualifying_short);
                case "SPRINT": return res.getString(R.string.race_sprint_short);
                case "QUALIFYING": return res.getString(R.string.race_qualifying_short);
                case "RACE": return res.getString(R.string.race_race_short);
                default: return name;
            }
        }
    }

    static final class Race {
        final int round;
        final String name;
        final String place;
        /** In start order; the last one is the race. */
        final List<Session> sessions;

        Race(int round, String name, String place, List<Session> sessions) {
            this.round = round;
            this.name = name;
            this.place = place;
            List<Session> sorted = new ArrayList<>(sessions);
            sorted.sort((a, b) -> a.start.compareTo(b.start));
            this.sessions = Collections.unmodifiableList(sorted);
        }

        /** "United States Grand Prix" becomes "United States GP", which fits the panel's top line. */
        String shortName() {
            return name.replace("Grand Prix", "GP").trim();
        }
    }

    /** What the panel shows at one moment of a race weekend. */
    static final class State {
        final Race race;
        /** One per session of {@link #race}, same order. */
        final List<Status> statuses;
        /** The session on track now, or null. */
        final Session live;
        /** The next session to start, or null once the race has started. */
        final Session next;
        /** Time until {@link #next} starts; null when there is no next session. */
        final Duration untilNext;

        State(Race race, List<Status> statuses, Session live, Session next, Duration untilNext) {
            this.race = race;
            this.statuses = Collections.unmodifiableList(statuses);
            this.live = live;
            this.next = next;
            this.untilNext = untilNext;
        }

        /** The race is over and the panel is only showing the weekend until {@link #TAIL} runs out. */
        boolean finished() {
            return live == null && next == null;
