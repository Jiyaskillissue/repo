     * "3 days ago, 9:40 pm", "Oct 12, 9:40 pm". The parts are Android's own wording in the TV's language.
     */
    static String describe(Context ctx, long when) {
        long now = System.currentTimeMillis();
        // Under a minute Android says "0 minutes ago"; one minute reads better.
        when = Math.min(when, now - DateUtils.MINUTE_IN_MILLIS);
        if (now - when < DateUtils.HOUR_IN_MILLIS) {
            return DateUtils.getRelativeTimeSpanString(when, now, DateUtils.MINUTE_IN_MILLIS).toString();
        }
        String day = DateUtils.getRelativeTimeSpanString(when, now, DateUtils.DAY_IN_MILLIS).toString();
        // It follows "Last opened" mid-sentence: "yesterday", not "Yesterday". Month names keep their capital.
        if (Locale.getDefault().getLanguage().equals("en") && (day.startsWith("Yesterday") || day.startsWith("Today"))) {
            day = Character.toLowerCase(day.charAt(0)) + day.substring(1);
        }
        return day + ", " + DateUtils.formatDateTime(ctx, when, DateUtils.FORMAT_SHOW_TIME);
    }

        return ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE);

import android.content.Intent;
import android.net.Uri;
import java.util.Map;
import java.util.function.BiFunction;

/** Selects a configured, package-scoped action only when it resolves; otherwise keeps the normal launch. */
final class LaunchActions {
    /** A launch_actions value "view:<uri>" opens that URI in the package instead of sending an action. */
    static final String VIEW_PREFIX = "view:";

    private LaunchActions() {}

    static <T> T select(String pkg, Map<String, String> actions, T fallback,
                        BiFunction<String, String, T> resolver) {
        String action = actions.get(pkg);
        if (action == null) return fallback;
        T custom = resolver.apply(pkg, action);
        return custom != null ? custom : fallback;
    }

    private static final String INPUT_PREFIX = "input:";

    /** The TV input id from an "input:<id>" launch_actions value, or null when the value isn't one, including
     *  an empty id ("input:" alone), which falls back to a normal launch rather than an empty input switch. */
    static String inputId(String action) {
        if (action == null || !action.startsWith(INPUT_PREFIX)) return null;
        String id = action.substring(INPUT_PREFIX.length()).trim();
        return id.isEmpty() ? null : id;
    }

    /** The URI of a "view:<uri>" value, or null for a plain action. */
    static String viewUri(String value) {
        if (value == null || !value.startsWith(VIEW_PREFIX)) return null;
        String uri = value.substring(VIEW_PREFIX.length()).trim();
        return uri.isEmpty() ? null : uri;
    }

    /** The intent for a launch_actions value, limited to its package: ACTION_VIEW on the URI, or the action. */
    static Intent intent(String pkg, String value) {
        String uri = viewUri(value);
        Intent i = uri != null ? new Intent(Intent.ACTION_VIEW, Uri.parse(uri)) : new Intent(value);
        return i.setPackage(pkg);
    }

import android.content.Context;

/**
 * locale: the language the launcher shows itself in, instead of the TV's. A China-market Xiaomi offers Chinese and
 * English only, so a Vietnamese owner saw the launcher in Chinese although it is translated into Vietnamese.
 * Each activity wraps its base context here; an empty or malformed tag leaves the TV's language.
 */
final class LauncherLocale {
    // Subtags after the language, so "sr-Latn" and "ar-EG-u-nu-latn" (Arabic with Western digits) pass too.
    private static final Pattern TAG = Pattern.compile("[a-z]{2,3}(-[A-Za-z0-9]{1,8})*");

    private LauncherLocale() {
    }

    /** The Locale for a config tag such as "vi", "pt-BR" or "ar-EG-u-nu-latn", or null for "" or anything that is not a plain tag. */
    static Locale parse(String tag) {
        if (tag == null || !TAG.matcher(tag.trim()).matches()) return null;
        return Locale.forLanguageTag(tag.trim());
    }

    /** base with the configured language, or base itself when locale is off. */
    static Context wrap(Context base) {
        Locale locale = parse(Config.load(base).locale);
        if (locale == null) return base;
        // Formatters that take no Context (the clock's date, the lunar line) follow the default locale.
        Locale.setDefault(locale);
        Configuration c = new Configuration(base.getResources().getConfiguration());
        c.setLocale(locale);
        return base.createConfigurationContext(c);
    }

import android.content.Context;

/** Home shortcut choices made from the library, separate from theme defaults and tile ordering. */
final class LibraryApps {
    static final int MAX_ADDED = 128;
    private static final long RECENT_MS = 7L * 24 * 60 * 60 * 1000;
    private static final int MAX_BYTES = MAX_ADDED * 256;

    private LibraryApps() { }

    private static AtomicFile store(Context context, boolean removed) {
        return new AtomicFile(new File(context.getFilesDir(), removed ? "library_home_removed.txt" : "library_apps.txt"));
    }

    static Set<String> load(Context context) {
        return read(store(context, false));
    }

    static Set<String> loadRemoved(Context context) { return read(store(context, true)); }

    private static Set<String> read(AtomicFile file) {
        try { return readStrict(file); }
        catch (IOException | RuntimeException e) {
            Log.w("TvHome.Library", "Could not read shortcuts", e);
            return Collections.emptySet();
        }
    }

    private static Set<String> readStrict(AtomicFile file) throws IOException {
        if (!file.getBaseFile().exists() && !new File(file.getBaseFile() + ".bak").exists())
            return Collections.emptySet();
        try (FileInputStream input = file.openRead()) {
            ByteArrayOutputStream bytes = new ByteArrayOutputStream();
            byte[] buffer = new byte[1024];
            int count;
            while ((count = input.read(buffer)) != -1) {
                if (bytes.size() + count > MAX_BYTES) throw new IOException("Shortcut file too large");
                bytes.write(buffer, 0, count);
            }
            return decode(new String(bytes.toByteArray(), StandardCharsets.UTF_8));
        }
    }

    static boolean save(Context context, String pkg, boolean onHome) {
        Set<String> additions, removed;
        try {
            additions = readStrict(store(context, false));
            removed = readStrict(store(context, true));
        } catch (IOException | RuntimeException e) {
            // Never replace unreadable owner choices with an empty list on the next edit.
            Log.w("TvHome.Library", "Keeping unreadable shortcut choices unchanged", e);
            return false;
        }
        if (onHome) {
            if (!write(store(context, false), additions, pkg, true)) return false;
            return !removed.contains(pkg) || write(store(context, true), removed, pkg, false);
        }
        // The removal takes precedence over theme defaults and any previous explicit addition.
        if (!write(store(context, true), removed, pkg, true)) return false;
        if (additions.contains(pkg)) write(store(context, false), additions, pkg, false);
        return true;
    }

    private static boolean write(AtomicFile file, Set<String> saved, String pkg, boolean added) {
        FileOutputStream output = null;
        try {
            Set<String> next = change(saved, pkg, added);
            byte[] bytes = encode(next).getBytes(StandardCharsets.UTF_8);
            output = file.startWrite();
            output.write(bytes);
            file.finishWrite(output);
            return true;
        } catch (IOException | RuntimeException e) {
            if (output != null) file.failWrite(output);
            Log.w("TvHome.Library", "Could not save shortcut choice", e);
            return false;
        }
    }

    static boolean visible(boolean tvApp, boolean themeHidden, boolean explicitlyAdded) {
        return visible(tvApp, themeHidden, explicitlyAdded, false);
    }

    static boolean visible(boolean tvApp, boolean themeHidden, boolean explicitlyAdded, boolean removedFromHome) {
        return !removedFromHome && (explicitlyAdded || (tvApp && !themeHidden));

    static boolean excludedFromCatalog(Set<String> excludedApps, String pkg) {

    static boolean okOpens(boolean selection, boolean openAll, boolean onHome) {
        return !selection && (openAll || onHome);

    static boolean recent(long installedAt, long now) {
        return installedAt > 0 && installedAt <= now && now - installedAt <= RECENT_MS;
    }

    /** Missing shortcuts first; recent installs lead that group. The caller sorts ties by label. */
    static int comparePriority(boolean homeA, long installedA, boolean homeB, long installedB, long now) {
        if (homeA != homeB) return homeA ? 1 : -1;
        if (homeA) return 0;
        boolean recentA = recent(installedA, now), recentB = recent(installedB, now);
        if (recentA != recentB) return recentA ? -1 : 1;
        return recentA ? Long.compare(installedB, installedA) : 0;
    }

    /** After adding a shortcut, prefer another missing app; otherwise stay by a neighbour. */
    static int nextFocus(int current, int count, IntPredicate missing) {
        if (count <= 0) return -1;
        current = Math.max(0, Math.min(current, count - 1));
        for (int step = 1; step < count; step++) {
            int candidate = (current + step) % count;
            if (missing.test(candidate)) return candidate;
        }
        return current + 1 < count ? current + 1 : Math.max(0, current - 1);
    }

    /** Checked Home apps first, then apps hidden from Home; the caller sorts each group by label. */
    static int compareHomeGroup(boolean homeA, boolean homeB) {
        return Boolean.compare(homeB, homeA);
    }

    /** Keep the next neighbour when a shortcut moves away, or the previous one at the end. */
    static int nextAfterRemoval(int current, int count) {
        if (count <= 1) return -1;
        current = Math.max(0, Math.min(current, count - 1));
        return current + 1 < count ? current + 1 : current - 1;
    }

    static int compareCatalog(boolean homeA, String labelA, String packageA,
                              boolean homeB, String labelB, String packageB, Collator collator) {
        int group = compareHomeGroup(homeA, homeB);
        if (group != 0) return group;
        int label = collator.compare(labelA, labelB);
        return label != 0 ? label : packageA.compareTo(packageB);
    }

    static Set<String> change(Set<String> saved, String pkg, boolean added) {
        if (!validPackage(pkg)) throw new IllegalArgumentException("Invalid package");
        Set<String> next = new HashSet<>(saved);
        if (added) next.add(pkg); else next.remove(pkg);
        if (next.size() > MAX_ADDED) throw new IllegalArgumentException("Too many shortcuts");
        return next;
    }

    static String encode(Set<String> added) {
        if (added.size() > MAX_ADDED) throw new IllegalArgumentException("Too many shortcuts");
        StringBuilder text = new StringBuilder();
        for (String pkg : new TreeSet<>(added)) {
            if (!validPackage(pkg)) throw new IllegalArgumentException("Invalid package");
            text.append(pkg).append('\n');
        }
        return text.toString();
    }

    static Set<String> decode(String text) {
        if (text.length() > MAX_BYTES) throw new IllegalArgumentException("Shortcut file too large");
        Set<String> result = new HashSet<>();
        int start = 0;
        for (int i = 0; i <= text.length(); i++) {
            if (i < text.length() && text.charAt(i) != '\n') continue;
            if (i == start && i == text.length()) break;
            String pkg = text.substring(start, i);
            if (!validPackage(pkg) || !result.add(pkg) || result.size() > MAX_ADDED)
                throw new IllegalArgumentException("Invalid shortcut file");
            start = i + 1;
        }
        return result;
    }

    static boolean validPackage(String pkg) {
        if (pkg == null || pkg.isEmpty() || pkg.length() > 255) return false;
        boolean first = true;
        for (int i = 0; i < pkg.length(); i++) {
            char c = pkg.charAt(i);
            if (c == '.') {
                if (first) return false;
                first = true;
            } else {
                boolean letter = c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c == '_';
                if (!letter && (first || c < '0' || c > '9')) return false;
                first = false;
            }
        }
        return !first;
    }
}

/**

 * and so does a package change (install, update, removal) while it is open or still loading; only the latest load

    private int latest;
    private boolean revealPending;

    /** Starts a load, from opening the library when reveal is true. Returns its number. */
    int start(boolean reveal) {
        if (reveal) revealPending = true;
        return ++latest;
    }

    /** Whether a finished load is the latest one, the only one that fills the catalogue. */
    boolean isLatest(int load) {
        return load == latest;
    }

    /** Once the latest load has filled the catalogue: whether the popup still has to appear. True once per opening. */
    boolean takeReveal() {
        boolean reveal = revealPending;
        revealPending = false;
        return reveal;
    }

final class Lunar {

    private Lunar() {

    /** Hours east of UTC used for the calendar: lunar_utc_offset, or 7 for Vietnamese, 9 for Korean and 8 otherwise. */
    static int offsetFor(int configured, Locale locale) {

            case "ko": return 9;

    /** Today's lunar date, formatted with the locale's strings. */
    static String today(Resources res, int utcOffset) {
        Calendar c = Calendar.getInstance();
        int[] l = fromSolar(c.get(Calendar.DAY_OF_MONTH), c.get(Calendar.MONTH) + 1, c.get(Calendar.YEAR), utcOffset);
        String month = l[3] == 1 ? res.getString(R.string.lunar_month_leap, l[1]) : String.valueOf(l[1]);
        String[] stems = res.getStringArray(R.array.lunar_stems);
        String[] branches = res.getStringArray(R.array.lunar_branches);
        String year = res.getString(R.string.lunar_year, stems[Math.floorMod(l[2] + 6, 10)], branches[Math.floorMod(l[2] + 8, 12)]);
        return res.getString(R.string.lunar_date_format, l[0], month, year);
    }

    /** Gregorian date to {day, month, year, leap (1 = leap month)} of the lunar calendar at UTC+tz. */
    static int[] fromSolar(int dd, int mm, int yy, double tz) {
        long dayNumber = jdFromDate(dd, mm, yy);
        int k = (int) Math.floor((dayNumber - 2415021.076998695) / 29.530588853) + 1;
        long monthStart = newMoonDay(k, tz);
        // The mean-month estimate can still be a new moon ahead when the real one falls just after local midnight;
        // one step back was not always enough, and the day before that new moon came out as day 0.
        while (monthStart > dayNumber) monthStart = newMoonDay(--k, tz);
        long a11 = lunarMonth11(yy, tz);
        long b11 = a11;
        int lunarYear;
        if (a11 >= monthStart) {
            lunarYear = yy;
            a11 = lunarMonth11(yy - 1, tz);
        } else {
            lunarYear = yy + 1;
            b11 = lunarMonth11(yy + 1, tz);
        }
        int lunarDay = (int) (dayNumber - monthStart + 1);
        int diff = (int) Math.floor((monthStart - a11) / 29.0);
        int leap = 0;
        int lunarMonth = diff + 11;
        if (b11 - a11 > 365) {
            int leapDiff = leapMonthOffset(a11, tz);
            if (diff >= leapDiff) {
                lunarMonth = diff + 10;
                if (diff == leapDiff) leap = 1;
            }
        }
        if (lunarMonth > 12) lunarMonth -= 12;
        if (lunarMonth >= 11 && diff < 4) lunarYear -= 1;
        return new int[]{lunarDay, lunarMonth, lunarYear, leap};
    }

    static long jdFromDate(int dd, int mm, int yy) {
        int a = (14 - mm) / 12;
        long y = yy + 4800 - a;
        int m = mm + 12 * a - 3;
        long jd = dd + (153 * m + 2) / 5 + 365 * y + y / 4 - y / 100 + y / 400 - 32045;
        if (jd < 2299161) jd = dd + (153 * m + 2) / 5 + 365 * y + y / 4 - 32083;
        return jd;
    }

    /** Day number of the k-th new moon after 1900-01-01, in local time. */
    static long newMoonDay(int k, double tz) {
        double t = k / 1236.85, t2 = t * t, t3 = t2 * t, dr = Math.PI / 180;
        double jd1 = 2415020.75933 + 29.53058868 * k + 0.0001178 * t2 - 0.000000155 * t3;
        jd1 += 0.00033 * Math.sin((166.56 + 132.87 * t - 0.009173 * t2) * dr);
        double m = 359.2242 + 29.10535608 * k - 0.0000333 * t2 - 0.00000347 * t3;
        double mpr = 306.0253 + 385.81691806 * k + 0.0107306 * t2 + 0.00001236 * t3;
        double f = 21.2964 + 390.67050646 * k - 0.0016528 * t2 - 0.00000239 * t3;
        double c1 = (0.1734 - 0.000393 * t) * Math.sin(m * dr) + 0.0021 * Math.sin(2 * dr * m);
        // Term by term as in Há»“ Ngá»c Äá»©c's code, each with its own sign: folding two terms into one "c1 -= a + b"
        // flips the second one, which moved new moons by hours and put the 2023 leap month a month early.
        c1 = c1 - 0.4068 * Math.sin(mpr * dr) + 0.0161 * Math.sin(dr * 2 * mpr);
        c1 = c1 - 0.0004 * Math.sin(dr * 3 * mpr);
        c1 = c1 + 0.0104 * Math.sin(dr * 2 * f) - 0.0051 * Math.sin(dr * (m + mpr));
        c1 = c1 - 0.0074 * Math.sin(dr * (m - mpr)) + 0.0004 * Math.sin(dr * (2 * f + m));
        c1 = c1 - 0.0004 * Math.sin(dr * (2 * f - m)) - 0.0006 * Math.sin(dr * (2 * f + mpr));
        c1 += 0.0010 * Math.sin(dr * (2 * f - mpr)) + 0.0005 * Math.sin(dr * (2 * mpr + m));
        double deltaT = t < -11
                ? 0.001 + 0.000839 * t + 0.0002261 * t2 - 0.00000845 * t3 - 0.000000081 * t * t3
                : -0.000278 + 0.000265 * t + 0.000262 * t2;
        return (long) Math.floor(jd1 + c1 - deltaT + 0.5 + tz / 24);
    }

    /** Which of the 12 thirty-degree sectors the sun's longitude is in at the start of the day. */
    static int sunLongitude(long jdn, double tz) {
        double t = (jdn - 2451545.5 - tz / 24) / 36525, t2 = t * t, dr = Math.PI / 180;
        double m = 357.52910 + 35999.05030 * t - 0.0001559 * t2 - 0.00000048 * t * t2;
        double l0 = 280.46645 + 36000.76983 * t + 0.0003032 * t2;
        double dl = (1.914600 - 0.004817 * t - 0.000014 * t2) * Math.sin(dr * m);
        dl += (0.019993 - 0.000101 * t) * Math.sin(dr * 2 * m) + 0.000290 * Math.sin(dr * 3 * m);
        double l = (l0 + dl) * dr;
        l -= Math.PI * 2 * Math.floor(l / (Math.PI * 2));
        return (int) Math.floor(l / Math.PI * 6);
    }

    /** First day of the 11th lunar month (the one holding the winter solstice) of a Gregorian year. */
    static long lunarMonth11(int yy, double tz) {
        long off = jdFromDate(31, 12, yy) - 2415021;
        int k = (int) Math.floor(off / 29.530588853);
        long nm = newMoonDay(k, tz);
        if (sunLongitude(nm, tz) >= 9) nm = newMoonDay(k - 1, tz);
        return nm;
    }

    /** Index of the leap month after month 11 in a 13-month year: the first month with no solar term. */
    static int leapMonthOffset(long a11, double tz) {
        int k = (int) Math.floor((a11 - 2415021.076998695) / 29.530588853 + 0.5);
        int i = 1;
        int arc = sunLongitude(newMoonDay(k + i, tz), tz);
        int last;
        do {
            last = arc;
            i++;
            arc = sunLongitude(newMoonDay(k + i, tz), tz);
        } while (arc != last && i < 14);
        return i - 1;
    }

            /**
             * Every "bring this tile on screen" scroll lands here, both RecyclerView's own on focus and
             * onTileFocused's. The stock version treats the grid's top padding as the visible top, but that
             * padding only pushes the dock down the screen: grid rows scroll up into it. With the stock rule,
             * Up into a row above the padding dragged that row down to the padding line, cut it off at the
             * bottom of the screen and could scroll the focused tile out entirely, dropping focus into the
             * top bar. Grid tiles now scroll just enough to show the rect between the grid's top edge and its
             * bottom; dock tiles are left to onTileFocused, which parks the dock row on the padding line.
             */
            @Override
            public boolean requestChildRectangleOnScreen(@NonNull RecyclerView parent, @NonNull View child,
                                                         @NonNull Rect rect, boolean immediate,
                                                         boolean focusedChildVisible) {
                if (singleRow) {
                    // Sideways: scroll just enough to bring the tile's cell inside the padding. Cells are a
                    // column wide, so the row moves a whole tile at a time and the tiles stay on the columns.
                    // The whole cell, not the card rect passed in: the card is narrower and centred in it.
                    int left = child.getLeft();
                    int right = child.getRight();
                    int dx;
                    if (left < getPaddingLeft()) dx = left - getPaddingLeft();
                    else if (right > getWidth() - getPaddingRight()) dx = right - (getWidth() - getPaddingRight());
                    else dx = 0;
                    if (dx == 0) return false;
                    if (immediate || reducedMotion || !animations) parent.scrollBy(dx, 0);
                    else parent.smoothScrollBy(dx, 0);
                    return true;
                }
                if (dockRow && getPosition(child) < COLUMNS) return false;
                int top = child.getTop() + rect.top;
                int bottom = top + rect.height();
                // A row already fully on screen stays put, even inside the bottom padding (the last row
                // rows_below_dock leaves visible ends there; nudging it up was the bug). A row that has to scroll
                // in still stops at the padding line, clear of the screen edge, as before.
                int limitBottom = getHeight() - getPaddingBottom();
                int dy;
                if (top < 0) dy = top;
