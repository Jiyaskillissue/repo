            jm = 1 + days / 31;
            jd = 1 + days % 31;

            jd = 1 + (days - 186) % 30;
        }

/**
 * Time-of-day sky palette for the optional "sunset" backdrop (config.xml, sky_intensity).
 *
 * Pure maths, no Android types and no clock of its own: the caller passes minutes since midnight,
 * so the palette can be tested on the JVM. The colours are interpolated between a handful of
 * anchor points through the day, which is what makes the backdrop drift from night to dawn to
 * sunset instead of switching at fixed hours. Every colour stays dark enough for the white
 * text, the hero name and the dock band to keep their contrast.
 */
final class Sky {

    /** One anchor: a minute of the day and the values the palette takes exactly there. */
    static final class Stop {
        final int minute;
        final int top;
        final int mid;
        final int bottom;
        /** Alpha of the sun/moon glow near the horizon, 0 to 255. */
        final int sun;
        /** Horizontal position of the glow, 0 = left edge, 1 = right edge. */
        final float sunX;

        Stop(int minute, int top, int mid, int bottom, int sun, float sunX) {
            this.minute = minute;
            this.top = top;
            this.mid = mid;
            this.bottom = bottom;
            this.sun = sun;
            this.sunX = sunX;
        }
    }

    /** The palette at one instant: the three gradient colours plus the glow. */
    static final class Palette {
        final int top;
        final int mid;
        final int bottom;
        final int sun;
        final float sunX;

        Palette(int top, int mid, int bottom, int sun, float sunX) {
            this.top = top;
            this.mid = mid;
            this.bottom = bottom;
            this.sun = sun;
            this.sunX = sunX;
        }
    }

    private static final int NIGHT_TOP = 0xFF04060E;
    private static final int NIGHT_MID = 0xFF070C1A;
    private static final int NIGHT_BOTTOM = 0xFF0B1326;

    // Anchors were picked around a temperate evening: full night until half past five, dawn before
    // seven, the warmest glow between seven in the evening and a quarter to eight. With a location
    // set, at(minute, sunrise, sunset) stretches this reference day onto the real one.
    static final int REF_SUNRISE = 450;
    static final int REF_SUNSET = 1140;
    private static final Stop[] STOPS = {
            new Stop(0, NIGHT_TOP, NIGHT_MID, NIGHT_BOTTOM, 0, 0.30f),
            new Stop(320, NIGHT_TOP, NIGHT_MID, NIGHT_BOTTOM, 0, 0.30f),

            new Stop(1260, 0xFF0A0A1C, 0xFF191230, 0xFF3A1F49, 60, 0.86f),
            new Stop(1350, NIGHT_TOP, NIGHT_MID, NIGHT_BOTTOM, 0, 0.30f),
            new Stop(1440, NIGHT_TOP, NIGHT_MID, NIGHT_BOTTOM, 0, 0.30f),
    };

    private Sky() {
    }

    /**
     * The palette at {@code minuteOfDay} on a day whose sun rises at {@code sunrise} and sets at
     * {@code sunset} (minutes of the day): night, day and evening are each stretched piecewise so
     * the reference sunrise and sunset glows land on the real ones.
     */
    static Palette at(int minuteOfDay, int sunrise, int sunset) {
        if (sunrise <= 0 || sunset >= 1440 || sunrise >= sunset) return at(minuteOfDay);
        int m = Math.max(0, Math.min(1440, minuteOfDay));
        float ref;
        if (m < sunrise) ref = m * REF_SUNRISE / (float) sunrise;
        else if (m < sunset) ref = REF_SUNRISE + (m - sunrise) * (REF_SUNSET - REF_SUNRISE) / (float) (sunset - sunrise);
        else ref = REF_SUNSET + (m - sunset) * (1440 - REF_SUNSET) / (float) (1440 - sunset);
        return at(Math.round(ref));
    }

    /** The palette at {@code minuteOfDay}, clamped to a day. */
    static Palette at(int minuteOfDay) {
        int m = Math.max(0, Math.min(1440, minuteOfDay));
        Stop a = STOPS[0];
        Stop b = STOPS[STOPS.length - 1];
        for (int i = 0; i < STOPS.length - 1; i++) {
            if (m >= STOPS[i].minute && m <= STOPS[i + 1].minute) {
                a = STOPS[i];
                b = STOPS[i + 1];
                break;
            }
        }
        float t = b.minute == a.minute ? 0f : (m - a.minute) / (float) (b.minute - a.minute);
        return new Palette(
                lerpColor(a.top, b.top, t),
                lerpColor(a.mid, b.mid, t),
                lerpColor(a.bottom, b.bottom, t),
                lerp(a.sun, b.sun, t),
                lerp(a.sunX, b.sunX, t));
    }

    /** Minute of day, 0 to 1439, of a wall clock time. */
    static int minuteOfDay(int hour, int minute) {
        return Math.max(0, Math.min(23, hour)) * 60 + Math.max(0, Math.min(59, minute));
    }

    /** Blends two opaque ARGB colours channel by channel; {@code t} is 0 to 1. */
    static int lerpColor(int from, int to, float t) {
        float f = Math.max(0f, Math.min(1f, t));
        int a = Math.round(((from >>> 24) & 0xFF) + (((to >>> 24) & 0xFF) - ((from >>> 24) & 0xFF)) * f);
        int r = Math.round(((from >> 16) & 0xFF) + (((to >> 16) & 0xFF) - ((from >> 16) & 0xFF)) * f);
        int g = Math.round(((from >> 8) & 0xFF) + (((to >> 8) & 0xFF) - ((from >> 8) & 0xFF)) * f);
        int b = Math.round((from & 0xFF) + ((to & 0xFF) - (from & 0xFF)) * f);
        return (a << 24) | (r << 16) | (g << 8) | b;
    }

    private static int lerp(int from, int to, float t) {
        return Math.round(from + (to - from) * Math.max(0f, Math.min(1f, t)));
    }

    private static float lerp(float from, float to, float t) {
        return from + (to - from) * Math.max(0f, Math.min(1f, t));
    }
}

package com.gathin.tvhome;

import java.time.LocalDate;
import java.time.ZoneId;

/**
 * Sunrise and sunset for a place, computed offline with NOAA's general solar position equations
 * (accurate to a few minutes, which is plenty for a backdrop). Pure maths, no Android types, so
 * it runs in JVM tests; no network and no location permission, the place comes from config.xml.
 */
final class Sun {

    private Sun() {
    }

    /**
     * Local minutes of the day of sunrise and sunset, as {@code {sunrise, sunset}}, or null when the
     * sun does not rise or does not set that day (polar night or midnight sun).
     */
    static int[] riseAndSet(LocalDate date, double latitude, double longitude, ZoneId zone) {
        double g = 2 * Math.PI / 365 * (date.getDayOfYear() - 1);
        double eqTime = 229.18 * (0.000075 + 0.001868 * Math.cos(g) - 0.032077 * Math.sin(g)
                - 0.014615 * Math.cos(2 * g) - 0.040849 * Math.sin(2 * g));
        double decl = 0.006918 - 0.399912 * Math.cos(g) + 0.070257 * Math.sin(g)
                - 0.006758 * Math.cos(2 * g) + 0.000907 * Math.sin(2 * g)
                - 0.002697 * Math.cos(3 * g) + 0.00148 * Math.sin(3 * g);
        double lat = Math.toRadians(latitude);
        // 90.833 degrees: the sun's upper edge on the horizon, with refraction.
        double cosHa = Math.cos(Math.toRadians(90.833)) / (Math.cos(lat) * Math.cos(decl))
                - Math.tan(lat) * Math.tan(decl);
        if (cosHa > 1 || cosHa < -1) return null;
        double ha = Math.toDegrees(Math.acos(cosHa));
        int offset = zone.getRules().getOffset(date.atTime(12, 0)).getTotalSeconds() / 60;
        return new int[]{
                (int) Math.round(720 - 4 * (longitude + ha) - eqTime + offset),
                (int) Math.round(720 - 4 * (longitude - ha) - eqTime + offset)};
    }
}

/** The files ThemeProvider may write or delete under Config.dir, by the URI's path segments. */
final class ThemeFiles {
    static final String WALLPAPERS = "wallpapers";
    /** The names tools/theme.sh accepts for a collection image, and the types WallpaperRotation reads. */
    private static final Pattern IMAGE = Pattern.compile("[A-Za-z0-9_.-]+\\.(jpg|jpeg|png|webp)",
            Pattern.CASE_INSENSITIVE);

    private ThemeFiles() {}

    /** The path relative to the theme directory, e.g. "config.json" or "wallpapers/dusk.jpg", or null if refused. */
    static String path(List<String> segments) {
        if (segments == null) return null;
        if (segments.size() == 1) {
            String name = segments.get(0);
            return Config.FILE_NAME.equals(name) || "wallpaper.jpg".equals(name) || "theme_revision".equals(name)
                    ? name : null;
        }
        if (segments.size() == 2 && WALLPAPERS.equals(segments.get(0))
                && (collectionImage(segments.get(1)) || WallpaperCaptions.FILE_NAME.equals(segments.get(1)))) {
            return WALLPAPERS + "/" + segments.get(1);
        }
        return null;
    }

    /** True for "wallpapers" alone: a delete of it clears the collection's images. */
    static boolean isCollection(List<String> segments) {
        return segments != null && segments.size() == 1 && WALLPAPERS.equals(segments.get(0));
    }

    static boolean collectionImage(String name) {
        return name != null && !name.startsWith(".") && IMAGE.matcher(name).matches();
    }

 * locked out of Android/data (TCL G07, Android 11). Only the files {@link ThemeFiles} allows, and only for
 * callers holding WRITE_SECURE_SETTINGS, which the adb shell has and ordinary apps cannot get.
 *
 *   content write --uri content://com.gathin.tvhome.theme/config.json < /data/local/tmp/config.json
 *   content write --uri content://com.gathin.tvhome.theme/wallpapers/dusk.jpg < /data/local/tmp/dusk.jpg
 *   content delete --uri content://com.gathin.tvhome.theme/config.json
 *   content delete --uri content://com.gathin.tvhome.theme/wallpapers      (every image in the collection)
 */
public class ThemeProvider extends ContentProvider {

    private File file(Uri uri) throws FileNotFoundException {
        String path = ThemeFiles.path(uri.getPathSegments());
        File dir = Config.dir(getContext());
        if (dir == null || path == null) throw new FileNotFoundException(String.valueOf(uri));
        File f = new File(dir, path);
        File parent = f.getParentFile();
        if (parent != null && !parent.isDirectory() && !parent.mkdirs()) throw new FileNotFoundException(String.valueOf(uri));
        return f;
    }

    @Override
    public ParcelFileDescriptor openFile(Uri uri, String mode) throws FileNotFoundException {
        File f = file(uri);
        if (mode.contains("w")) {
            return ParcelFileDescriptor.open(f, ParcelFileDescriptor.MODE_WRITE_ONLY
                    | ParcelFileDescriptor.MODE_CREATE | ParcelFileDescriptor.MODE_TRUNCATE);
        }
        return ParcelFileDescriptor.open(f, ParcelFileDescriptor.MODE_READ_ONLY);
    }

    @Override
    public int delete(Uri uri, String selection, String[] selectionArgs) {
        File dir = Config.dir(getContext());

        }

            return file(uri).delete() ? 1 : 0;
        } catch (FileNotFoundException e) {

        }

package com.gathin.tvhome;

/**
 * Whether Home rebuilds its tiles when it comes back to the front. A rebuild asks PackageManager for every app again
 * and reloads every banner, icon and label (about 30 ms on a TCL C755, on every Home press), so it happens only when
 * an app was installed, removed, updated, enabled or disabled in the meantime. Otherwise Home only puts the focus
 * back, as before.
 */
final class TileRefresh {

    enum Action { NOTHING, REFOCUS, REBUILD }

    private boolean started;
    /** The tiles may not match the installed apps: true until the first build. */
    private boolean stale = true;
    private boolean refocus;

    void onStart() {
        started = true;
    }

    /** Another app took the screen: its tiles stay, and focus goes back to them on the way in. */
    void onStop() {
        started = false;
        refocus = true;
    }

    /** Home pressed while Home is in front: back to the first tile. */
    void onHomeAgain() {
        refocus = true;
    }

    /** The tiles were dropped (onTrimMemory): build them again on the way back. */
    void invalidate() {
        stale = true;
    }

    /**
     * A package broadcast. True when Home is on screen and should rebuild now; while it is in the background the
     * rebuild waits for {@link #onResume}, so a run of app updates costs one rebuild instead of one each.
     */
    boolean onPackagesChanged() {
        if (started) return true;
        stale = true;
        return false;
    }

    /** @param changedMeanwhile PackageManager reports changed packages since the last build (a missed broadcast). */
    Action onResume(boolean changedMeanwhile) {
        Action action = stale || changedMeanwhile ? Action.REBUILD : refocus ? Action.REFOCUS : Action.NOTHING;
        stale = false;
        refocus = false;
        return action;
    }
}

import android.graphics.Canvas;
import android.graphics.LinearGradient;
import android.graphics.Paint;
import android.graphics.PorterDuff;
import android.graphics.PorterDuffXfermode;
import android.graphics.Shader;
import android.view.View;

import androidx.annotation.NonNull;
import androidx.recyclerview.widget.RecyclerView;

/**
 * fade_rows: the rows fade out as they scroll up under the clock, instead of being cut off at the grid's top
 * edge. It erases the top of the grid's own picture, so whatever is behind Home (a wallpaper, the sky) shows
 * through; the grid draws into a hardware layer for that ({@link #attach}), or the erasing would punch through
 * the whole window. Android's own requiresFadingEdge draws its fade in the wrong place with this grid's
 * clipChildren=false, so it is not used. The fade grows in over the first pixels of scrolling, so the dock at
 * rest is untouched.
 */
final class TopFade extends RecyclerView.ItemDecoration {

    private final float length;
    private final Paint paint = new Paint();
    private float shaderHeight = -1f;

    private TopFade(float length) {
        this.length = length;
        paint.setXfermode(new PorterDuffXfermode(PorterDuff.Mode.DST_OUT));
    }

    static void attach(RecyclerView grid, float length) {
        grid.setLayerType(View.LAYER_TYPE_HARDWARE, null);
        grid.addItemDecoration(new TopFade(length));
    }

    /** How tall the fade is after scrolling {@code scrolled} pixels: nothing at rest, then up to {@code length}. */
    static float height(int scrolled, float length) {
        return Math.max(0f, Math.min(length, scrolled));
    }

    @Override
    public void onDrawOver(@NonNull Canvas c, @NonNull RecyclerView parent, @NonNull RecyclerView.State state) {
        float fade = height(parent.computeVerticalScrollOffset(), length);
        if (fade <= 0f) return;
        if (fade != shaderHeight) {
            paint.setShader(new LinearGradient(0f, 0f, 0f, fade, 0xFF000000, 0x00000000, Shader.TileMode.CLAMP));
            shaderHeight = fade;
        }
        c.drawRect(0f, 0f, parent.getWidth(), fade, paint);
    }
}
